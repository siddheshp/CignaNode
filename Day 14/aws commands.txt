# Set environment variables
$env:AWS_REGION = 'us-east-1'
$env:AWS_ACCOUNT_ID = '797721580785'
$env:ECR_REPOSITORY = 'ems-backend'
$env:IMAGE_TAG = 'v1'
$env:ECR_REGISTRY = "$env:AWS_ACCOUNT_ID.dkr.ecr.$env:AWS_REGION.amazonaws.com"
$env:IMAGE_NAME = "$env:ECR_REGISTRY/$env:ECR_REPOSITORY"

# Login to ECR
aws ecr get-login-password --region $env:AWS_REGION | podman login --username AWS --password-stdin $env:ECR_REGISTRY

# Push images
podman push "$($env:IMAGE_NAME):$($env:IMAGE_TAG)"
podman push "$($env:IMAGE_NAME):latest"
